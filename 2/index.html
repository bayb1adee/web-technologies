Iterating an array with forEach()

The forEach operator walks through every element of an array.

Demo

We have an array of words. Direct shows them in order; Shifted shows each word moved one position forward (a rotation).

Press a button to render the array.

Code

var a = ["Alpha", "Beta", "Gamma", "Delta"];

// Regular (non-arrow) functions — refactored to arrows in Project 3.
function First() {
  var box = document.querySelector("#cards");
  box.innerHTML = "";
  a.forEach(function (item) {
    box.innerHTML += "<div>" + item + "</div>";
  });
}

function Second() {
  var box = document.querySelector("#cards");
  box.innerHTML = "";
  a.forEach(function (item, index) {
    var shifted = a[(index + 1) % a.length];
    box.innerHTML += "<div>" + shifted + "</div>";
  });
}
