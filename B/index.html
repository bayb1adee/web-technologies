ToDo List

An object-oriented, event-driven ToDo list.

Demo

The Add button is disabled until you type something. Click a task's × to remove it.

To Do

    Code (object-oriented, ES6)

    class ToDo {
  constructor(list, input, submit) {
    this.list = list;
    this.input = input;
    this.submit = submit;

    // By default, the submit button is disabled.
    this.submit.disabled = true;

    // Enable the button only if there is text in the input field.
    this.input.onkeyup = () => {
      this.submit.disabled = this.input.value.length === 0;
    };

    this.form = this.input.closest("form");
    this.form.onsubmit = () => this.add();
  }

  add() {
    const li = document.createElement("li");
    li.innerHTML = this.input.value;
    const x = document.createElement("span");
    x.className = "x";
    x.textContent = "×";
    x.onclick = () => li.remove();
    li.appendChild(x);
    this.list.append(li);

    this.input.value = "";          // clear the input field
    this.submit.disabled = true;    // disable again
    return false;                   // stop the form from submitting
  }
}

document.addEventListener("DOMContentLoaded", () => {
  new ToDo(
    document.querySelector("#tasks"),
    document.querySelector("#task"),
    document.querySelector("#submit")
  );
});
