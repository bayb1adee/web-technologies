Code

function quote(id, abbr) {
  var u = "https://api.nbrb.by/exrates/rates/" + id;
  $.getJSON(u)
    .done(function (data) {
      $("#price").text(data.Cur_OfficialRate + " (" + data.Cur_Abbreviation + ")");
      $("#status").text("Updated: " + data.Date);
    })
    .fail(function () {
      $("#status").text("Could not load the rate (check the connection).");
    });
}
